{{/* Override of nicokaiser/hugo-theme-gallery's get-gallery.html. For filtered pages (`type: filtered`), the page's own bundle holds only the cover image, so the upstream count of `len $images` reports "1 photo" on every tag-album card. Replace that count — both for the page itself and inside the parent-section aggregation — with the number of images that actually match the filter. */}} {{ $gallery := "" }} {{ $images := where (.Resources.ByType "image") "Params.hidden" "ne" true }} {{ if gt (len $images) 0 }} {{ $covers := where (.Resources.ByType "image") "Params.cover" "eq" true }} {{ $cover := "" }} {{ if ge (len $covers) 0 }} {{ $cover = index $covers 0 }} {{ end }} {{ $featured := $cover | default ((.Resources.ByType "image").GetMatch (.Params.featured_image | default "*feature*")) | default (index $images 0) }} {{ $thumbnail := $featured.Filter (slice images.AutoOrient (images.Process "fit 600x600")) }} {{ $color := index $thumbnail.Colors 0 | default "transparent" }} {{ $imageCount := 0 }} {{ $albumCount := 0 }} {{ if eq .Type "filtered" }} {{ $imageCount = partial "filtered-image-count.html" . }} {{ else if .IsPage }} {{ $imageCount = len $images }} {{ else }} {{ range where .RegularPagesRecursive "Params.private" "ne" true }} {{ $albumCount = add $albumCount 1 }} {{ if eq .Type "filtered" }} {{ $imageCount = add $imageCount (partial "filtered-image-count.html" .) }} {{ else }} {{ $imageCount = add $imageCount (len (where (.Resources.ByType "image") "Params.hidden" "ne" true)) }} {{ end }} {{ end }} {{ end }} {{ $gallery = dict "page" $ "images" $images "thumbnail" $thumbnail "color" $color "albumCount" $albumCount "imageCount" $imageCount }} {{ end }} {{ return $gallery }}
